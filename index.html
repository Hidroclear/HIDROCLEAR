function doPost(e) {
  try {

        ]]);

      var filaNum = hojaM.getLastRow() + 1;
      hojaM.getRange(filaNum, 1, 1, 10).setValues([[
        fechaObj, datos.tipo, datos.codigo, datos.cantidad,
        datos.ubicacion, datos.ubicacion_destino || '',
        datos.responsable, docTexto, datos.despacho || '', 'PENDIENTE'
      ]]);
      hojaM.getRange(filaNum, 1).setNumberFormat('dd/mm/yyyy');
      hojaM.getRange(filaNum, 8).setNumberFormat('@');
      return ok({ mensaje: 'Movimiento registrado correctamente' });
    }

    if (tipoPeticion === 'MARCAR_PROCESADO') {
      var hojaC = ss.getSheetByName('COLA_MOVIMIENTOS');
      if (hojaC) hojaC.getRange(datos.fila, 10).setValue('PROCESADO');
      return ok({});
    }

    var mapHojas = { INVENTARIO: 'Hoja 1', CODIGOS: 'CODIGOS', UBICACIONES: 'UBICACIONES' };
    var nombreHoja = mapHojas[tipoPeticion] || 'Hoja 1';
    var hoja = ss.getSheetByName(nombreHoja) || ss.insertSheet(nombreHoja);
    if (datos.bloque === 0) hoja.clearContents();
    if (datos.valores && datos.valores.length > 0) {
      hoja.getRange(hoja.getLastRow() + 1, 1, datos.valores.length, datos.valores[0].length)

function doGet(e) {

    var ss = SpreadsheetApp.openById('18ABHh-28OwyxIKTtehCgjFlbW1PG1B8zLR6sBXYCwJw');
    var tipo = e.parameter.tipo || 'INVENTARIO';

          documento: String(filas[j][7] || '').trim(),
          despacho: String(filas[j][8] || '').trim()
        });
      }
      return ok({ datos: res });

function leerInventario(ss) {
  var h = ss.getSheetByName('Hoja 1');
  if (!h || h.getLastRow() < 2) return [];
  var f = h.getDataRange().getValues();
  var r = [];
  for (var i = 1; i < f.length; i++) {
    if (!f[i][0]) continue;
    r.push({
      codigo: String(f[i][0]),
      descripcion: String(f[i][1] || ''),
      stock: Number(f[i][2]) || 0,
      bodega: String(f[i][3] || ''),
      categoria: String(f[i][4] || '')
    });
  }
  return r;
}

function leerCodigos(ss) {
  var h = ss.getSheetByName('CODIGOS');
  if (!h || h.getLastRow() < 2) return [];
  var f = h.getDataRange().getValues();
  var r = [];
  for (var i = 1; i < f.length; i++) {
    if (!f[i][0]) continue;
    r.push({
      codigo: String(f[i][0]).trim(),
      descripcion: String(f[i][1] || '').trim(),
      categoria: String(f[i][2] || '').trim()
    });
  }
  return r;
}

function leerUbicaciones(ss) {
  var h = ss.getSheetByName('UBICACIONES');
  if (!h || h.getLastRow() < 2) return [];
  var f = h.getDataRange().getValues();
  var r = [];
  for (var i = 1; i < f.length; i++) {
    if (!f[i][0]) continue;
    r.push({ ubicacion: String(f[i][0]).trim() });
  }
  return r;
}

function parseFecha(str) {
  if (!str) return new Date();
  str = String(str).trim();
  if (/^\d{2}\/\d{2}\/\d{4}$/.test(str)) {
    var p = str.split('/');
    return new Date(parseInt(p[2]), parseInt(p[1])-1, parseInt(p[0]));
  }
  if (/^\d{4}-\d{2}-\d{2}$/.test(str)) {
    var q = str.split('-');
    return new Date(parseInt(q[0]), parseInt(q[1])-1, parseInt(q[2]));
  }
  var d = new Date(str);
  return isNaN(d.getTime()) ? new Date() : d;
}

function ok(obj) {
  obj.status = 'ok';
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function error(msg) {
  return ContentService.createTextOutput(JSON.stringify({ status: 'error', mensaje: msg }))
    .setMimeType(ContentService.MimeType.JSON);
}
